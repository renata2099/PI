USE SistemaChamados;
GO

SELECT 
    dp.name AS UserName,
    dp.sid AS UserSID,
    sp.name AS LoginName,
    sp.sid AS LoginSID,
    CASE 
        WHEN sp.sid IS NULL THEN 'ORFAO - Sem login no servidor!'
        WHEN dp.sid <> sp.sid THEN 'SID DIFERENTE - Conflito!'
        ELSE 'OK'
    END AS Status
FROM sys.database_principals dp
LEFT JOIN master.sys.server_principals sp ON dp.name = sp.name
WHERE dp.type IN ('S', 'U')  -- SQL Users e Windows Users
  AND dp.name NOT IN ('public', 'guest', 'INFORMATION_SCHEMA', 'sys', 'dbo')
  AND dp.name LIKE '%pi_grp%';
