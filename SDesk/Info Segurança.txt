-- Ver todos os schemas que ainda pertencem ao pi_grp1
SELECT 
    s.name AS SchemaName,
    s.schema_id,
    dp.name AS OwnerName,
    dp.principal_id
FROM sys.schemas s
JOIN sys.database_principals dp ON s.principal_id = dp.principal_id
WHERE dp.name = 'pi_grp1';

-- Ver se ele é membro de alguma role
SELECT 
    r.name AS RoleName,
    m.name AS MemberName
FROM sys.database_role_members rm
JOIN sys.database_principals r ON rm.role_principal_id = r.principal_id
JOIN sys.database_principals m ON rm.member_principal_id = m.principal_id
WHERE m.name = 'pi_grp1';
